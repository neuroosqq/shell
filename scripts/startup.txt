# Startup script for shell emulator
# Lines starting with # are comments and are skipped.

# Show current directory listing
ls

# Show current path
pwd

# Try to change to a subdirectory
cd docs

# Show listing inside subdirectory
ls

# Show current path
pwd

# Search for a file in current subtree
find about.txt

# Go back to parent
cd ..

# Search for a file from the root
find hello.py

# Search for a missing file
find missing.txt

# Try a command with a quoted argument
cd "my folder"

# Unknown command
unknowncommand

# Show history of all commands
history

# Exit
exit
